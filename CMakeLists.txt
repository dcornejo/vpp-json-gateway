cmake_minimum_required(VERSION 3.22)
project(vpp_json_gateway VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WITH_VAPI "Build the VPP 26.06 adapter (Linux)" OFF)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HIREDIS REQUIRED IMPORTED_TARGET hiredis)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)
find_package(nlohmann_json 3.10 REQUIRED)
add_library(gateway_core src/common.cc src/schema.cc src/backend.cc src/transfer.cc src/store.cc)
target_include_directories(gateway_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(gateway_core PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto PkgConfig::SQLITE Threads::Threads)
target_compile_options(gateway_core PUBLIC -Wall -Wextra -Wpedantic -fno-exceptions)
target_compile_definitions(gateway_core PUBLIC JSON_NOEXCEPTION)
add_executable(vpp-schema-generator tools/schema_generator.cc src/schema.cc src/common.cc)
target_include_directories(vpp-schema-generator PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(vpp-schema-generator PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto)
target_compile_definitions(vpp-schema-generator PRIVATE JSON_NOEXCEPTION)
target_compile_options(vpp-schema-generator PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
if(WITH_VAPI)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "VAPI requires Linux; target VPP 26.06 / Ubuntu 26.04")
  endif()
  find_path(VAPI_INCLUDE_DIR vapi/interface.api.vapi.h REQUIRED)
  find_library(VAPI_LIBRARY NAMES vapiclient REQUIRED)
  find_path(VPP_API_DIR interface.api.json PATH_SUFFIXES share/vpp/api/core REQUIRED)
  set(VPP_API_MODULES "interface;vlib;ip" CACHE STRING "VPP API modules to expose")
  list(APPEND VPP_API_MODULES interface)
  list(REMOVE_DUPLICATES VPP_API_MODULES)
  set(VPP_API_INPUTS)
  foreach(module IN LISTS VPP_API_MODULES)
    list(APPEND VPP_API_INPUTS "${VPP_API_DIR}/${module}.api.json")
  endforeach()
  set(GENERATED_API "${CMAKE_CURRENT_BINARY_DIR}/generated_api.cc")
  add_custom_command(OUTPUT "${GENERATED_API}"
    COMMAND vpp-schema-generator "${GENERATED_API}" ${VPP_API_INPUTS}
    DEPENDS vpp-schema-generator ${VPP_API_INPUTS} VERBATIM)
  target_sources(gateway_core PRIVATE src/vapi_backend.cc "${GENERATED_API}")
  target_include_directories(gateway_core SYSTEM PRIVATE ${VAPI_INCLUDE_DIR})
  target_link_libraries(gateway_core PUBLIC ${VAPI_LIBRARY})
  target_compile_definitions(gateway_core PUBLIC WITH_VAPI)
endif()
add_executable(vpp-json-gateway src/main.cc src/redis_transport.cc)
target_link_libraries(vpp-json-gateway PRIVATE gateway_core PkgConfig::HIREDIS)
add_executable(gateway-client tools/client.cc src/redis_transport.cc)
target_link_libraries(gateway-client PRIVATE gateway_core PkgConfig::HIREDIS)
add_executable(example-client example/client.cc src/redis_transport.cc)
target_link_libraries(example-client PRIVATE gateway_core PkgConfig::HIREDIS)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.cc)
  target_link_libraries(core_test PRIVATE gateway_core)
  add_test(NAME core COMMAND core_test)
  add_executable(schema_test tests/schema_test.cc)
  target_link_libraries(schema_test PRIVATE gateway_core)
  add_test(NAME schema COMMAND schema_test)
  add_executable(large_message_test tests/large_message_test.cc)
  target_link_libraries(large_message_test PRIVATE gateway_core)
  add_test(NAME large_message COMMAND large_message_test)
  if(WITH_VAPI AND "vlib" IN_LIST VPP_API_MODULES AND "ip" IN_LIST VPP_API_MODULES)
    add_executable(generated_reply_test tests/generated_reply_test.cc)
    set_source_files_properties(tests/generated_reply_test.cc PROPERTIES OBJECT_DEPENDS "${GENERATED_API}")
    target_include_directories(generated_reply_test PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_include_directories(generated_reply_test SYSTEM PRIVATE ${VAPI_INCLUDE_DIR})
    target_link_libraries(generated_reply_test PRIVATE gateway_core)
    add_test(NAME generated_reply COMMAND generated_reply_test)
  endif()
endif()
install(TARGETS vpp-json-gateway gateway-client RUNTIME DESTINATION bin)

install(FILES LICENSE NOTICE DESTINATION share/doc/vpp-json-gateway)
